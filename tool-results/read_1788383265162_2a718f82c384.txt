    30→function countryFlag(code: string): string {
    31→  if (!/^[A-Z]{2}$/.test(code)) return "🌐"
    32→  return String.fromCodePoint(...[...code].map((c) => 127397 + c.charCodeAt(0)))
    33→}
    34→
    35→function fmtNum(n: number): string {
    36→  if (n >= 1_000_000) return (n / 1_000_000).toFixed(1) + "M"
    37→  if (n >= 1_000) return (n / 1_000).toFixed(1) + "k"
    38→  return String(Math.round(n))
    39→}
    40→
    41→function fmtDuration(s: number): string {
    42→  if (!s) return "0s"
    43→  const m = Math.floor(s / 60)
    44→  const sec = Math.round(s % 60)
    45→  return m ? `${m}m ${sec}s` : `${sec}s`
    46→}
