   197→/** Generic list editor with collapsible cards + reorder + remove + add */
   198→function ListEditor<T>({
   199→  items,
   200→  onChange,
   201→  createItem,
   202→  renderFields,
   203→  itemTitle,
   204→  max = 12,
   205→  addLabel = "Add item",
   206→  label,
   207→}: {
   208→  items: T[]
   209→  onChange: (items: T[]) => void
   210→  createItem: () => T
   211→  renderFields: (item: T, update: (patch: Partial<T>) => void) => React.ReactNode
   212→  itemTitle: (item: T, i: number) => string
   213→  max?: number
   214→  addLabel?: string
   215→  label?: string
   216→}) {
   217→  const [open, setOpen] = React.useState<number | null>(null)
   218→  const updateAt = (i: number, patch: Partial<T>) => {
   219→    const next = items.slice()
   220→    next[i] = { ...next[i], ...patch }
   221→    onChange(next)
   222→  }
   223→  const move = (i: number, dir: -1 | 1) => {
   224→    const j = i + dir
   225→    if (j < 0 || j >= items.length) return
   226→    const next = items.slice()
   227→    const [x] = next.splice(i, 1)
   228→    next.splice(j, 0, x)
   229→    onChange(next)
   230→    setOpen(j)
   231→  }
   232→  return (
   233→    <div className="space-y-2">
   234→      {label && <Label className="text-[11px] font-semibold uppercase tracking-wider text-zinc-400">{label} ({items.length})</Label>}
   235→      {items.map((item, i) => (
   236→        <div key={i} className="overflow-hidden rounded-lg border border-zinc-800 bg-zinc-900/40">
   237→          <div className="flex items-center gap-1 px-2 py-1.5">
   238→            <button
   239→              type="button"
   240→              className="flex flex-1 items-center gap-2 truncate text-left"
   241→              onClick={() => setOpen(open === i ? null : i)}
   242→            >
   243→              <ChevronsUpDown className={cn("h-3.5 w-3.5 shrink-0 text-zinc-500 transition-transform", open === i && "rotate-180")} />
   244→              <span className="truncate text-[12px] text-zinc-200">{itemTitle(item, i)}</span>
   245→            </button>
   246→            <Button variant="ghost" size="icon" className="h-6 w-6 text-zinc-500 hover:text-zinc-200" onClick={() => move(i, -1)} aria-label="Move up" disabled={i === 0}>
   247→              <ArrowUp className="h-3 w-3" />
   248→            </Button>
   249→            <Button variant="ghost" size="icon" className="h-6 w-6 text-zinc-500 hover:text-zinc-200" onClick={() => move(i, 1)} aria-label="Move down" disabled={i === items.length - 1}>
   250→              <ArrowDown className="h-3 w-3" />
   251→            </Button>
   252→            <Button
   253→              variant="ghost"
   254→              size="icon"
   255→              className="h-6 w-6 text-zinc-500 hover:text-rose-300"
   256→              onClick={() => onChange(items.filter((_, idx) => idx !== i))}
   257→              aria-label="Remove item"
   258→            >
   259→              <Trash2 className="h-3 w-3" />
   260→            </Button>
   261→          </div>
   262→          {open === i && <div className="space-y-2.5 border-t border-zinc-800 p-2.5">{renderFields(item, (patch) => updateAt(i, patch))}</div>}
   263→        </div>
   264→      ))}
   265→      {items.length < max && (
   266→        <Button
   267→          variant="outline"
   268→          size="sm"
   269→          className="h-7 w-full border-dashed border-zinc-700 bg-transparent text-[11px] text-zinc-400 hover:border-violet-500/50 hover:text-violet-300"
   270→          onClick={() => {
   271→            onChange([...items, createItem()])
   272→            setOpen(items.length)
   273→          }}
   274→        >
   275→          <Plus className="mr-1 h-3 w-3" /> {addLabel}
   276→        </Button>
   277→      )}
   278→    </div>
   279→  )
   280→}

   282→/** Simple list of editable strings (companies, field labels…) */
   283→function StringListEditor({ label, items, onChange, createValue, addLabel, max = 12, placeholder }: { label?: string; items: string[]; onChange: (items: string[]) => void; createValue: () => string; addLabel?: string; max?: number; placeholder?: string }) {
   284→  const updateAt = (i: number, v: string) => {
   285→    const next = items.slice()
   286→    next[i] = v
   287→    onChange(next)
   288→  }
   289→  return (
   290→    <div className="space-y-2">
   291→      {label && <Label className="text-[11px] font-semibold uppercase tracking-wider text-zinc-400">{label} ({items.length})</Label>}
   292→      {items.map((item, i) => (
   293→        <div key={i} className="flex items-center gap-1">
   294→          <Input
   295→            value={item}
   296→            placeholder={placeholder}
   297→            onChange={(e) => updateAt(i, e.target.value)}
   298→            className="h-8 border-zinc-700/80 bg-zinc-900/60 text-[13px] text-zinc-100"
   299→          />
   300→          <Button variant="ghost" size="icon" className="h-7 w-7 shrink-0 text-zinc-500 hover:text-rose-300" onClick={() => onChange(items.filter((_, idx) => idx !== i))} aria-label="Remove">
   301→            <Trash2 className="h-3 w-3" />
   302→          </Button>
   303→        </div>
   304→      ))}
   305→      {items.length < max && (
   306→        <Button
   307→          variant="outline"
   308→          size="sm"
   309→          className="h-7 w-full border-dashed border-zinc-700 bg-transparent text-[11px] text-zinc-400 hover:border-violet-500/50 hover:text-violet-300"
   310→          onClick={() => onChange([...items, createValue()])}
   311→        >
   312→          <Plus className="mr-1 h-3 w-3" /> {addLabel ?? "Add"}
   313→        </Button>
   314→      )}
   315→    </div>
   316→  )

   844→function SectionEditor({ section }: { section: Section }) {
   845→  const update = (patch: Partial<Section>) => {
   846→    useForge.getState().updateSection(section.id, patch as Partial<Section>)
   847→  }
   848→  switch (section.type) {
   849→    case "navbar":
   850→      return <NavbarEditor section={section} update={update as unknown as (p: Partial<NavbarSection>) => void} />
   851→    case "hero":
   852→      return <HeroEditor section={section} update={update as unknown as (p: Partial<HeroSection>) => void} />
   853→    case "logos":
   854→      return <LogosEditor section={section} update={update as unknown as (p: Partial<LogosSection>) => void} />
   855→    case "features":
   856→      return <FeaturesEditor section={section} update={update as unknown as (p: Partial<FeaturesSection>) => void} />
   857→    case "stats":
   858→      return <StatsEditor section={section} update={update as unknown as (p: Partial<StatsSection>) => void} />
   859→    case "testimonials":
   860→      return <TestimonialsEditor section={section} update={update as unknown as (p: Partial<TestimonialsSection>) => void} />
   861→    case "pricing":
   862→      return <PricingEditor section={section} update={update as unknown as (p: Partial<PricingSection>) => void} />
   863→    case "faq":
   864→      return <FaqEditor section={section} update={update as unknown as (p: Partial<FaqSection>) => void} />
   865→    case "gallery":
   866→      return <GalleryEditor section={section} update={update as unknown as (p: Partial<GallerySection>) => void} />
   867→    case "contact":
   868→      return <ContactEditor section={section} update={update as unknown as (p: Partial<ContactSection>) => void} />
   869→    case "cta-final":
   870→      return <CtaFinalEditor section={section} update={update as unknown as (p: Partial<CtaFinalSection>) => void} />
   871→    case "footer":
   872→      return <FooterEditor section={section} update={update as unknown as (p: Partial<FooterSection>) => void} />
