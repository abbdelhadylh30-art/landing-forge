    32→function countryFlag(code: string): string {
    33→  if (!/^[A-Z]{2}$/.test(code)) return "🌐"
    34→  return String.fromCodePoint(...[...code].map((c) => 127397 + c.charCodeAt(0)))
    35→}

    37→function fmtNum(n: number): string {
    38→  if (n >= 1_000_000) return (n / 1_000_000).toFixed(1) + "M"
    39→  if (n >= 1_000) return (n / 1_000).toFixed(1) + "k"
    40→  return String(Math.round(n))
    41→}
    42→
    43→function fmtDuration(s: number): string {
    44→  if (!s) return "0s"
    45→  const m = Math.floor(s / 60)
    46→  const sec = Math.round(s % 60)
    47→  return m ? `${m}m ${sec}s` : `${sec}s`
    48→}

    51→  if (d === "mobile") return Smartphone

    55→}

    57→function fmtLiveTimer(s: number): string {
    58→  if (s < 60) return `${s}s`
    59→  const m = Math.floor(s / 60)
    60→  return `${m}m ${String(s % 60).padStart(2, "0")}s`
    61→}

   226→export function DashboardView() {
   227→  const projectId = useForge((s) => s.project.id)
   228→  const projectName = useForge((s) => s.project.name)
   229→  const slug = useForge((s) => s.project.slug)
   230→  const updateSection = useForge((s) => s.updateSection)
   231→  const sections = useForge((s) => s.config.sections)
   232→
   233→  const [data, setData] = React.useState<AnalyticsPayload | null>(null)
   234→  const [leads, setLeads] = React.useState<LeadRecord[]>([])
   235→  const [selectedLead, setSelectedLead] = React.useState<LeadRecord | null>(null)
   236→  const [loading, setLoading] = React.useState(true)
   237→  const [seeding, setSeeding] = React.useState(false)
   238→  const [days, setDays] = React.useState("30")
   239→  // live refresh: poll while enabled + tab visible; only re-render on change
   240→  const [live, setLive] = React.useState(true)
   241→  const [lastRefresh, setLastRefresh] = React.useState<number | null>(null)
   242→  const [hasNew, setHasNew] = React.useState(false)
   243→
   244→  // ── real-time push relay (WebSocket) — presence + instant event signals.
   245→  // Falls back silently to REST polling whenever the relay is unreachable.
   246→  const relay = useDashboardRelay(projectId, live)
   247→  const pushConnected = live && relay.connected
   248→
   249→  const load = React.useCallback(
   250→    async (opts?: { quiet?: boolean }) => {
   251→      if (!projectId) return
   252→      if (!opts?.quiet) setLoading(true)
   253→      try {
   254→        const res = await fetch(`/api/analytics?projectId=${projectId}&days=${days}`)
   255→        const payload = (await res.json()) as AnalyticsPayload
   256→        setData((prev) => {
   257→          if (prev && JSON.stringify(prev) === JSON.stringify(payload)) return prev
   258→          if (prev && opts?.quiet) setHasNew(true)
   259→          return payload
   260→        })
   261→      } catch {
   262→        if (!opts?.quiet) toast.error("Could not load analytics")
   263→      } finally {
   264→        if (!opts?.quiet) setLoading(false)
   265→        if (opts?.quiet) setLastRefresh(Date.now())
   266→      }
   267→      try {
   268→        const res = await fetch(`/api/leads?projectId=${projectId}&take=50`)
   269→        const out = (await res.json()) as { leads?: LeadRecord[] }
   270→        setLeads((prev) => (prev && JSON.stringify(prev) === JSON.stringify(out.leads ?? []) ? prev : (out.leads ?? [])))
   271→      } catch {
   272→        /* leads inbox is best-effort */
   273→      }
   274→    },
   275→    [projectId, days]
   276→  )
   277→
   278→  React.useEffect(() => {
   279→    void load()
   280→  }, [load])
   281→
   282→  // ── Live polling: REST backstop while enabled + tab visible.
   283→  // While the push relay is connected we poll slowly (20s consistency check);
   284→  // without it we fall back to the original 5s cadence. Payloads are diffed —
   285→  // unchanged data never re-renders the charts.
   286→  React.useEffect(() => {
   287→    if (!live || !projectId) return
   288→    const t = setInterval(() => {
   289→      if (!document.hidden) void load({ quiet: true })
   290→    }, pushConnected ? 20_000 : 5_000)
   291→    return () => clearInterval(t)
   292→  }, [live, projectId, load, pushConnected])
   293→
   294→  // ── Push-driven refresh: relay signals (new visit / CTA click / form submit)
   295→  // trigger a quiet reload within ~1.2s — charts catch up without waiting for
   296→  // the polling backstop. Debounced so bursts collapse into one fetch.
   297→  React.useEffect(() => {
   298→    if (!relay.signals || !live) return
   299→    const t = setTimeout(() => {
   300→      if (!document.hidden) void load({ quiet: true })
   301→    }, 1200)
   302→    return () => clearTimeout(t)
   303→  }, [relay.signals, live, load])
   304→
   305→  const seed = async () => {
   306→    if (!projectId) return
   307→    setSeeding(true)
   308→    try {
   309→      const res = await fetch("/api/analytics/seed", {
   310→        method: "POST",
   311→        headers: { "content-type": "application/json" },
   312→        body: JSON.stringify({ projectId, days: Number(days) }),
   313→      })
   314→      const out = (await res.json()) as { pageviews?: number; events?: number; leads?: number; error?: string }
   315→      if (!res.ok) throw new Error(out.error)
   316→      toast.success("Demo traffic generated 🌍", { description: `${fmtNum(out.pageviews ?? 0)} pageviews · ${fmtNum(out.events ?? 0)} events · ${out.leads ?? 0} leads` })
   317→      await load()
   318→    } catch (e) {
   319→      toast.error("Seeding failed", { description: e instanceof Error ? e.message : undefined })
   320→    } finally {
   321→      setSeeding(false)
   322→    }
   323→  }
   324→
   325→  // clear the "new data" hint once the user has seen it (any interaction)
   326→  React.useEffect(() => {
   327→    if (!hasNew) return
   328→    const t = setTimeout(() => setHasNew(false), 6000)
   329→    return () => clearTimeout(t)
   330→  }, [hasNew])
   331→
   332→  const promoteWinner = async () => {
   333→    const hero = sections.find((s) => s.type === "hero" && s.ab?.enabled)

   335→    const winnerVariant = data.ab.variants.find((v) => v.name === data.ab!.winner)
   336→    if (!winnerVariant) return
   337→    updateSection(hero.id, {
   338→      headline: winnerVariant.headline,
   339→      ab: { ...hero.ab!, enabled: false },
   340→    })
   341→    if (projectId) {
   342→      // persist the promoted config so analytics & reloads agree (await before refetch)
   343→      const updated = useForge.getState().config
   344→      try {
   345→        await fetch(`/api/projects/${projectId}`, {
   346→          method: "PATCH",
   347→          headers: { "content-type": "application/json" },
   348→          body: JSON.stringify({ name: useForge.getState().project.name, config: updated }),
   349→        })
   350→      } catch {
   351→        /* keep local update even if persist fails */
   352→      }
   353→      void fetch("/api/analytics/track", {
   354→        method: "POST",
   355→        headers: { "content-type": "application/json" },
   356→        body: JSON.stringify({ projectId, type: "promote_winner", label: `Promoted variant ${data.ab.winner}`, variant: data.ab.winner }),
   357→      })
   358→    }
   359→    toast.success(`Variant ${data.ab.winner} promoted 👑`, { description: "Winning headline applied to the hero; test paused." })

   363→  const mergedLive = React.useMemo(() => {

   365→    if (!base) return null
   366→    // when the push relay is connected, its presence + leave-decisions are

   369→    return { ...base, active: mergeLiveVisits(base.active, relay.visits, left) }
   370→  }, [data, relay.visits, relay.connected, relay.leftIds])
   371→
   372→  if (loading && !data) {
   373→    return (
   374→      <div className="flex min-h-0 flex-1 items-center justify-center">
   375→        <div className="flex flex-col items-center gap-3 text-zinc-500">
   376→          <Loader2 className="h-6 w-6 animate-spin text-violet-400" />
   377→          <p className="text-[12px]">Loading analytics…</p>
   378→        </div>
   379→      </div>
   380→    )
   381→  }
   382→
   383→  const stats = data?.stats
   384→  const hasData = (stats?.pageviews ?? 0) > 0
   385→  const funnelMax = Math.max(1, ...(data?.funnel.map((f) => f.count) ?? [1]))
   386→
   387→  return (
   388→    <div className="lf-scroll min-h-0 flex-1 overflow-y-auto bg-zinc-950">
   389→      <div className="mx-auto max-w-6xl space-y-4 p-4 pb-16 sm:p-6">
   390→        {/* Header */}
   391→        <div className="flex flex-wrap items-center gap-3">
   392→          <div>
   393→            <h2 className="flex items-center gap-2 text-lg font-bold text-zinc-50">
   394→              <BarChart3 className="h-5 w-5 text-violet-300" /> Analytics — {projectName}
   395→            </h2>
   396→            <p className="text-[11px] text-zinc-500">
   397→              Privacy-friendly · no cookies · GDPR-safe · <span className="font-mono">/{slug}/dashboard</span>
   398→            </p>
   399→          </div>
   400→          <div className="ml-auto flex flex-wrap items-center gap-2">
   401→            {/* Live auto-refresh toggle — push relay w/ REST polling backstop */}
   402→            <button
   403→              type="button"
   404→              role="switch"
   405→              aria-checked={live}
   406→              onClick={() => setLive((v) => !v)}
   407→              title={
   408→                live
   409→                  ? pushConnected
   410→                    ? "Live push connected — visits & events stream over WebSocket instantly"
   411→                    : `Live refresh on (polling) — new published-page visits appear automatically${lastRefresh ? ` (last check ${Math.max(0, Math.round((Date.now() - lastRefresh) / 1000))}s ago)` : ""}`
   412→                  : "Live refresh paused — click to resume auto-updating"
   413→              }
   414→              className={cn(
   415→                "flex h-8 items-center gap-1.5 rounded-lg border px-2.5 text-[11px] font-semibold transition-colors",
   416→                live
   417→                  ? pushConnected
   418→                    ? "border-violet-500/40 bg-violet-500/10 text-violet-200 hover:bg-violet-500/15"

   427→                  <span className="relative flex size-2" aria-hidden>
