    31→function countryFlag(code: string): string {
    32→  if (!/^[A-Z]{2}$/.test(code)) return "🌐"
    33→  return String.fromCodePoint(...[...code].map((c) => 127397 + c.charCodeAt(0)))
    34→}
    35→
    36→function fmtNum(n: number): string {
    37→  if (n >= 1_000_000) return (n / 1_000_000).toFixed(1) + "M"
    38→  if (n >= 1_000) return (n / 1_000).toFixed(1) + "k"
    39→  return String(Math.round(n))
    40→}
    41→
    42→function fmtDuration(s: number): string {
    43→  if (!s) return "0s"
    44→  const m = Math.floor(s / 60)
    45→  const sec = Math.round(s % 60)
    46→  return m ? `${m}m ${sec}s` : `${sec}s`
    47→}
