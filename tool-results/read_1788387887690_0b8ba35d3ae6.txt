    31→function countryFlag(code: string): string {
    32→  if (!/^[A-Z]{2}$/.test(code)) return "🌐"
    33→  return String.fromCodePoint(...[...code].map((c) => 127397 + c.charCodeAt(0)))
    34→}
    35→
    36→function fmtNum(n: number): string {
    37→  if (n >= 1_000_000) return (n / 1_000_000).toFixed(1) + "M"
    38→  if (n >= 1_000) return (n / 1_000).toFixed(1) + "k"
    39→  return String(Math.round(n))
    40→}
    41→
    42→function fmtDuration(s: number): string {
    43→  if (!s) return "0s"
    44→  const m = Math.floor(s / 60)
    45→  const sec = Math.round(s % 60)
    46→  return m ? `${m}m ${sec}s` : `${sec}s`
    47→}
    48→
    49→function deviceIcon(d: string) {
    50→  if (d === "mobile") return Smartphone
    51→  if (d === "tablet") return Tablet
    52→  if (d === "desktop") return Monitor
    53→  return MonitorSmartphone
    54→}

    56→function fmtLiveTimer(s: number): string {
    57→  if (s < 60) return `${s}s`
    58→  const m = Math.floor(s / 60)
    59→  return `${m}m ${String(s % 60).padStart(2, "0")}s`
    60→}

   222→  const [days, setDays] = React.useState("30")

   224→  const [live, setLive] = React.useState(true)
   225→  const [lastRefresh, setLastRefresh] = React.useState<number | null>(null)
   226→  const [hasNew, setHasNew] = React.useState(false)
   227→
   228→  const load = React.useCallback(
   229→    async (opts?: { quiet?: boolean }) => {
   230→      if (!projectId) return
   231→      if (!opts?.quiet) setLoading(true)
   232→      try {
   233→        const res = await fetch(`/api/analytics?projectId=${projectId}&days=${days}`)
   234→        const payload = (await res.json()) as AnalyticsPayload
   235→        setData((prev) => {
   236→          if (prev && JSON.stringify(prev) === JSON.stringify(payload)) return prev
   237→          if (prev && opts?.quiet) setHasNew(true)
   238→          return payload
   239→        })
   240→      } catch {
   241→        if (!opts?.quiet) toast.error("Could not load analytics")
   242→      } finally {
   243→        if (!opts?.quiet) setLoading(false)

   246→      try {
   247→        const res = await fetch(`/api/leads?projectId=${projectId}&take=50`)
   248→        const out = (await res.json()) as { leads?: LeadRecord[] }
   249→        setLeads((prev) => (prev && JSON.stringify(prev) === JSON.stringify(out.leads ?? []) ? prev : (out.leads ?? [])))
   250→      } catch {
   251→        /* leads inbox is best-effort */
   252→      }
